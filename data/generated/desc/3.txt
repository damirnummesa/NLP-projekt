def solve(data_gdf: gpd.GeoDataFrame, europe_gdf: gpd.GeoDataFrame):
	# Filter the data to include only south EU countries (NUTS_0)
	south_eu_gdf = data_gdf[data_gdf['NUTS_0'].str.contains('ES|FR|IT|PT')]
	# Select columns of interest for further processing
	columns_of_interest = ['pH_H2O', 'Depth']
	transformed_data = south_eu_gdf[columns_of_interest].copy()
	# Group the data by Depth and calculate average pH_H2O
	average_pH_by_depth = transformed_data.groupby('Depth')['pH_H2O'].mean().reset_index()
	# Rename columns for clarity
	average_pH_by_depth.columns = ['depth', 'avg_ph_h2o']
	# Create a new GeoDataFrame with the calculated averages
	transformed_gdf = pd.DataFrame({'geometry': south_eu_gdf.geometry, 
	                                'Depth': average_pH_by_depth['depth'], 
	                                'Average pH_H2O': average_pH_by_depth['avg_ph_h2o']})
	# Convert transformed_data to a GeoDataFrame for consistency with the original data
	transformed_gdf = gpd.GeoDataFrame(transformed_gdf, geometry='geometry')
	import matplotlib.pyplot as plt
	transformed_gdf['Average pH_H2O'].plot(kind='hist', bins=10, figsize=(8,6))
	plt.title('Distribution of Average pH in South EU')
	plt.xlabel('pH Value')
	plt.ylabel('Frequency')
	plt.show()
