def solve(data_gdf: gpd.GeoDataFrame, europe_gdf: gpd.GeoDataFrame):
	# Filter the data to only include rows where LC0_Desc is not null and OC (20-30 cm) is greater than 2
	transformed_gdf = data_gdf[(data_gdf['LC0_Desc'].notnull()) & 
	                            ((data_gdf['OC (20-30 cm)'] > 2).astype(bool))]
	# Group the filtered data by LC0_Desc and calculate the average OC for each group
	average_OC_by_LC0_Desc = transformed_gdf.groupby('LC0_Desc')['OC'].mean().reset_index()
	# Rename the 'OC' column to 'Average_OC'
	average_OC_by_LC0_Desc.rename(columns={'OC': 'Average_OC'}, inplace=True)
	import matplotlib.pyplot as plt
	transformed_gdf.groupby('LC0_Desc')['OC'].mean().plot(kind='bar')
	plt.title('Average OC by Land Type (LC0_Desc)')
	plt.xlabel('Land Type (LC0_Desc)')
	plt.ylabel('Average OC')
	plt.xticks(rotation=90)
	plt.tight_layout()
	plt.savefig('average_oc_by_land_type.png', bbox_inches='tight')
