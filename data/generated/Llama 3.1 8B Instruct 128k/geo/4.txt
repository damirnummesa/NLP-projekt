def solve(data_gdf: gpd.GeoDataFrame, europe_gdf: gpd.GeoDataFrame):
	from sklearn.cluster import KMeans
	transformed_gdf = data_gdf.copy()
	kmeans = KMeans(n_clusters=3)
	kmeans.fit(data_gdf[['TH_LAT', 'TH_LONG']])
	data_gdf['cluster'] = kmeans.predict(data_gdf[['TH_LAT', 'TH_LONG']])
	fig, ax = plt.subplots(figsize=(10, 8))
	europe_gdf.plot(ax=ax, edgecolor='black', color='lightgray')
	transformed_gdf['cluster'] = KMeans(n_clusters=3).fit_predict(transformed_gdf[['TH_LAT', 'TH_LONG']])
	ax.scatter(transformed_gdf['TH_LAT'], transformed_gdf['TH_LONG'], c=transformed_gdf['cluster'])
	plt.savefig('kmeans_map.png')
