def solve(data_gdf: gpd.GeoDataFrame, europe_gdf: gpd.GeoDataFrame):
	import pandas as pd
	transformed_gdf = data_gdf[data_gdf['K'] > data_gdf['K'].median()]
	transformed_gdf['SURVEY_DATE'] = transformed_gdf['SURVEY_DATE'].astype('object')
	transformed_gdf.to_file('k_above_median.shp', driver='ESRI Shapefile')
	import matplotlib.pyplot as plt
	fig, ax = plt.subplots(figsize=(10, 8))
	europe_gdf.plot(ax=ax, edgecolor='black', color='lightgray')
	above_median_k = transformed_gdf[transformed_gdf['K'] > transformed_gdf['K'].median()]
	above_median_k.plot(ax=ax, column='K', markersize=5)
	ax.set_title('Locations where potassium (K) levels are above the median')
	ax.set_xlabel('Longitude')
	ax.set_ylabel('Latitude')
	plt.savefig('potassium_above_median.png')
	plt.show()
