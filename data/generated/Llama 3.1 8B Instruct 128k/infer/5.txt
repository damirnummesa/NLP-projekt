def solve(data_gdf: gpd.GeoDataFrame, europe_gdf: gpd.GeoDataFrame):
	# Filter the data to include only rows where 'K' is not missing and within a specific range (0-10000 mg/kg)
	data_gdf = data_gdf.dropna(subset=['K'])
	data_gdf = data_gdf[(data_gdf['K'] >= 10) & (data_gdf['K'] <= 10000)]
	# Select the columns of interest for linear regression analysis
	selected_columns = ['pH_H2O', 'K']
	transformed_gdf = data_gdf[selected_columns]
	# Perform a simple linear regression between 'pH_H2O' and 'K'
	from sklearn.linear_model import LinearRegression
	X = transformed_gdf['K'].values.reshape(-1, 1)
	y = transformed_gdf['pH_H2O']
	model = LinearRegression()
	model.fit(X, y)
	# Store the coefficients in a new GeoDataFrame
	transformed_gdf['slope'] = model.coef_[0]
	transformed_gdf['intercept'] = model.intercept_
	import matplotlib.pyplot as plt
	plt.figure(figsize=(10,6))
	sns.regplot(x='pH_H2O', y='K', data=transformed_gdf)
	plt.title('Linear Regression Analysis: pH_H2O vs K')
	plt.xlabel('pH of Water (pH_H2O)')
	plt.ylabel('Potassium Concentration (K)')
	plt.show()
