def solve(data_gdf: gpd.GeoDataFrame, europe_gdf: gpd.GeoDataFrame):
	transformed_data = data_gdf.loc[data_gdf['LC0_Desc'] != 'Non-classified'].copy()
	transformed_data = transformed_data.groupby('LC0_Desc')['K'].sum().reset_index(name='Total_K')
	fig, ax = plt.subplots(figsize=(10, 8))
	europe_gdf.plot(ax=ax, edgecolor='black', color='lightgray')
	transformed_data.plot(ax=ax, markersize=10)
	ax.set_title('Map with Data Points')
	ax.set_xlabel('Longitude')
	ax.set_ylabel('Latitude')
	plt.show()
