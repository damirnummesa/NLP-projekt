def solve(data_gdf: gpd.GeoDataFrame, europe_gdf: gpd.GeoDataFrame):
	transformed_data = data_gdf[['geometry', 'K']].copy()
	scaler = MinMaxScaler()
	transformed_K = scaler.fit_transform(transformed_data['K'].values.reshape(-1, 1))
	transformed_data['trans_K'] = transformed_K.flatten()
	fig, ax = plt.subplots(figsize=(10, 8))
	europe_gdf.plot(ax=ax, edgecolor='black', color='lightgray')
	transformed_data['K'].plot.hist(bins=50, alpha=0.5, ax=ax)
	transformed_data['trans_K'].plot.kde(ax=ax)
	ax.set_title('Histogram of K Values with Kernel Density Estimate')
	ax.set_xlabel('K Value')
	ax.set_ylabel('Frequency/KDE')
	plt.legend(['histogram', 'kde'])
	fig.savefig('k_kde.png')
